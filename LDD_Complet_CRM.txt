-- ============================================================================
-- 1. PROMOTION
-- ============================================================================
CREATE TABLE PROMOTION(
   id_promotion SERIAL,
   nom_promotion VARCHAR(100) NOT NULL,
   annee_diplome INTEGER NOT NULL,
   filiere VARCHAR(100) NOT NULL,
   PRIMARY KEY(id_promotion)
);

-- ============================================================================
-- 2. ENTREPRISE
-- ============================================================================
CREATE TABLE ENTREPRISE(
   id_entreprise SERIAL,
   nom_entreprise VARCHAR(150) NOT NULL,
   secteur_activite VARCHAR(150) NOT NULL,
   pays VARCHAR(100) NOT NULL,
   ville VARCHAR(100) NOT NULL,
   PRIMARY KEY(id_entreprise)
);

-- ============================================================================
-- 3. CERTIFICATION
-- ============================================================================
CREATE TABLE CERTIFICATION(
   id_certification SERIAL,
   nom_certification VARCHAR(150) NOT NULL,
   organisme VARCHAR(150) NOT NULL,
   PRIMARY KEY(id_certification)
);

-- ============================================================================
-- 4. AUDIT_LOG
-- ============================================================================
CREATE TABLE AUDIT_LOG(
   id_log SERIAL,
   action VARCHAR(100) NOT NULL,
   details TEXT,
   rows_affected INTEGER DEFAULT 0,
   executed_at TIMESTAMP DEFAULT NOW(),
   acteur VARCHAR(255) DEFAULT NULL,
   PRIMARY KEY(id_log)
);

-- ============================================================================
-- 5. QUESTIONNAIRE
-- ============================================================================
CREATE TABLE QUESTIONNAIRE(
   id_questionnaire SERIAL,
   titre VARCHAR(255) NOT NULL,
   description TEXT DEFAULT '',
   date_creation DATE DEFAULT CURRENT_DATE,
   actif BOOLEAN DEFAULT TRUE,
   PRIMARY KEY(id_questionnaire)
);

-- ============================================================================
-- 6. QUESTION
-- ============================================================================
CREATE TABLE QUESTION(
   id_question SERIAL,
   texte TEXT NOT NULL,
   type VARCHAR(50) NOT NULL DEFAULT 'text',
   options JSONB DEFAULT '[]',
   ordre INTEGER NOT NULL DEFAULT 0,
   tag VARCHAR(100) DEFAULT NULL,
   conditionnee_statut_emploi BOOLEAN NOT NULL DEFAULT FALSE,
   id_questionnaire INTEGER NOT NULL,
   PRIMARY KEY(id_question),
   FOREIGN KEY(id_questionnaire) REFERENCES QUESTIONNAIRE(id_questionnaire) ON DELETE CASCADE
);

-- ============================================================================
-- 7. OTP_CODES
-- ============================================================================
CREATE TABLE otp_codes(
   id SERIAL,
   email TEXT NOT NULL,
   code_hash TEXT NOT NULL,
   attempts INTEGER DEFAULT 0,
   created_at TIMESTAMP DEFAULT NOW(),
   expires_at TIMESTAMP NOT NULL,
   used BOOLEAN DEFAULT FALSE,
   ip_address TEXT DEFAULT '',
   PRIMARY KEY(id),
   UNIQUE(email)
);

-- ============================================================================
-- 8. SCHEMA_MIGRATIONS
-- ============================================================================
CREATE TABLE SCHEMA_MIGRATIONS(
   nom_fichier VARCHAR(255),
   date_execution TIMESTAMP NOT NULL DEFAULT NOW(),
   PRIMARY KEY(nom_fichier)
);

-- ============================================================================
-- 9. DEMANDE_RGPD
-- ============================================================================
CREATE TABLE DEMANDE_RGPD(
   id_demande SERIAL,
   type_demande VARCHAR(20) NOT NULL,
   statut VARCHAR(20) NOT NULL DEFAULT 'envoyee',
   date_demande TIMESTAMP NOT NULL DEFAULT NOW(),
   date_traitement TIMESTAMP,
   traitee_par VARCHAR(255),
   motif_refus TEXT,
   nom_complet VARCHAR(255),
   email VARCHAR(255),
   prise_en_charge_par VARCHAR(255),
   date_prise_en_charge TIMESTAMP,
   id_etudiant INTEGER,
   PRIMARY KEY(id_demande),
   CONSTRAINT chk_demande_type   CHECK (type_demande IN ('export', 'suppression')),
   CONSTRAINT chk_demande_statut CHECK (statut IN ('envoyee', 'en_traitement', 'traitee', 'rejetee'))
);

-- ============================================================================
-- 10. ETUDIANT
-- ============================================================================
CREATE TABLE ETUDIANT(
   id_etudiant SERIAL,
   nom VARCHAR(100) NOT NULL,
   prenom VARCHAR(100) NOT NULL,
   email VARCHAR(150) NOT NULL,
   email_academique VARCHAR(150),
   telephone VARCHAR(20) NOT NULL,
   date_naissance DATE NOT NULL,
   parcours_anterieur TEXT NOT NULL,
   date_inscription DATE NOT NULL,
   address VARCHAR(255) DEFAULT '',
   city VARCHAR(100) DEFAULT '',
   country VARCHAR(100) DEFAULT '',
   linkedin VARCHAR(255) DEFAULT '',
   availability_status VARCHAR(50) DEFAULT '',
   skills JSONB DEFAULT '[]'::jsonb,
   date_anonymisation TIMESTAMP DEFAULT NULL,
   id_promotion INTEGER NOT NULL,
   PRIMARY KEY(id_etudiant),
   CONSTRAINT etudiant_email_key            UNIQUE (email),
   CONSTRAINT etudiant_email_academique_key UNIQUE (email_academique)
);

-- ============================================================================
-- 11. EXPERIENCE_PRO
-- ============================================================================
CREATE TABLE EXPERIENCE_PRO(
   id_experience SERIAL,
   intitule_poste VARCHAR(150) NOT NULL,
   type_contrat VARCHAR(50) NOT NULL,
   date_debut DATE NOT NULL,
   date_fin DATE,
   salaire NUMERIC NOT NULL,

   poste_actuel BOOLEAN DEFAULT TRUE,
   id_entreprise INTEGER NOT NULL,

   PRIMARY KEY(id_experience)

-- ============================================================================
-- 12. OBTIENT

-- ============================================================================
-- 13. CONSENTEMENT_RGPD
-- ============================================================================
